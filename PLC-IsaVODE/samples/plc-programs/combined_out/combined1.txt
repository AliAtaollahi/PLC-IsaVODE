===== 3-IS_LOWER.ST =====
FUNCTION_BLOCK IS_LOWER

VAR_INPUT
	STR :	STRING;
END_VAR

VAR_OUTPUT
	IS_LOWER :	BOOL;
END_VAR

VAR
	pos :	INT;
	I :	INT;
END_VAR

I := LEN(str);
FOR pos := 1 TO I DO
	IF NOT ISC_LOWER(INT_TO_BYTE(GET_CHAR(str,pos))) THEN
		IS_LOWER := FALSE;
		RETURN;
	END_IF;
END_FOR;
IS_LOWER := I>0;

END_FUNCTION_BLOCK


===== 4-COUNT_CHAR.ST =====
FUNCTION_BLOCK COUNT_CHAR
	VAR_INPUT
		STR :	STRING;
		CHR :	BYTE;
	END_VAR

	VAR_OUTPUT
		COUNT_CHAR :	INT;
	END_VAR

	VAR
		I :	INT;
		pos :	INT;
	END_VAR

	I := LEN(str);
	COUNT_CHAR := 0;
	FOR pos := 1 TO I DO
		IF INT_TO_BYTE(GET_CHAR(str,pos)) = CHR THEN COUNT_CHAR := COUNT_CHAR + 1; END_IF;
	END_FOR;
END_FUNCTION_BLOCK
